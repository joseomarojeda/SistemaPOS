# Copia TODO este bloque y pegalo en PowerShell (clic derecho > Terminal), luego Enter.
$dir = "$env:USERPROFILE\Documents\SistemaPOS"
Get-ChildItem $dir -Recurse | Unblock-File
winget install --id OpenJS.NodeJS.LTS -e --accept-source-agreements --accept-package-agreements
$node = "$env:ProgramFiles\nodejs\node.exe"
if (-not (Test-Path $node)) { $node = (Get-Command node -ErrorAction SilentlyContinue).Source }
$ws = New-Object -ComObject WScript.Shell
foreach ($d in @([Environment]::GetFolderPath('Desktop'), [Environment]::GetFolderPath('Startup'))) {
  $l = $ws.CreateShortcut("$d\Punto de Venta.lnk")
  $l.TargetPath = $node
  $l.Arguments = '--disable-warning=ExperimentalWarning server.js --abrir'
  $l.WorkingDirectory = $dir
  $l.IconLocation = "$env:SystemRoot\System32\shell32.dll,43"
  $l.WindowStyle = 7
  $l.Save()
}
Start-Process powershell -Verb RunAs -ArgumentList '-NoProfile -Command "New-NetFirewallRule -DisplayName PuntoDeVenta -Direction Inbound -Protocol TCP -LocalPort 3000 -Action Allow"'
Start-Process $node -ArgumentList '--disable-warning=ExperimentalWarning server.js --abrir' -WorkingDirectory $dir
