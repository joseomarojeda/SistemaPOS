$dir = "$env:USERPROFILE\Documents\SistemaPOS"
Get-ChildItem $dir -Recurse | Unblock-File
winget install --id OpenJS.NodeJS.LTS -e --accept-source-agreements --accept-package-agreements
$ws = New-Object -ComObject WScript.Shell
$desktop = [Environment]::GetFolderPath('Desktop')
$l = $ws.CreateShortcut("$desktop\Punto de Venta.lnk")
$l.TargetPath = "$dir\INICIAR.bat"
$l.WorkingDirectory = $dir
$l.IconLocation = "$env:SystemRoot\System32\shell32.dll,43"
$l.WindowStyle = 7
$l.Save()
$auto = Join-Path ([Environment]::GetFolderPath('Startup')) 'Punto de Venta.lnk'
if (Test-Path -LiteralPath $auto) { Remove-Item -LiteralPath $auto -Force }
Start-Process powershell -Verb RunAs -ArgumentList '-NoProfile -Command "Get-NetFirewallRule -DisplayName PuntoDeVenta -ErrorAction SilentlyContinue | Remove-NetFirewallRule; New-NetFirewallRule -DisplayName PuntoDeVenta -Direction Inbound -Protocol TCP -LocalPort 3000,3001 -Action Allow"'
Write-Host 'Instalación lista. Inicia el sistema desde el icono Punto de Venta del escritorio.'
